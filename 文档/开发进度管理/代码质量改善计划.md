### 5.3 Hooks 提取方案

```typescript
// hooks/useChartInteraction.ts
export function useChartInteraction(options: {
  initialZoom?: number;
  initialPan?: { x: number; y: number };
  onZoomChange?: (zoom: number) => void;
}) {
  const [zoom, setZoom] = useState(options.initialZoom ?? 1);
  const [pan, setPan] = useState(options.initialPan ?? { x: 0, y: 0 });
  const [isDragging, setIsDragging] = useState(false);

  // 滚轮缩放
  const handleWheel = useCallback((e: WheelEvent) => { ... }, []);
  
  // 拖拽平移
  const handleDragStart = useCallback((e: MouseEvent) => { ... }, []);
  const handleDrag = useCallback((e: MouseEvent) => { ... }, []);
  const handleDragEnd = useCallback(() => { ... }, []);

  // 触摸手势
  const handleTouchStart = useCallback((e: TouchEvent) => { ... }, []);
  const handleTouchMove = useCallback((e: TouchEvent) => { ... }, []);
  const handleTouchEnd = useCallback(() => { ... }, []);

  return {
    zoom, setZoom,
    pan, setPan,
    isDragging,
    handlers: {
      onWheel: handleWheel,
      onMouseDown: handleDragStart,
      onMouseMove: handleDrag,
      onMouseUp: handleDragEnd,
      onTouchStart: handleTouchStart,
      onTouchMove: handleTouchMove,
      onTouchEnd: handleTouchEnd,
    },
  };
}

// 使用
function ChromaticityPlot() {
  const { zoom, pan, handlers } = useChartInteraction({
    initialZoom: 5,
    initialPan: { x: trueSource.labA, y: trueSource.labB },
  });
  
  return (
    <div {...handlers}>
      {/* 图表内容 */}
    </div>
  );
}
```

### 5.4 AuditLog 模型方案

```prisma
// 添加到 schema.prisma

model AuditLog {
  id          String   @id @default(cuid())
  
  // 操作信息
  action      AuditAction                    // 操作类型
  resource    String                         // 资源类型 (color, recipe, etc.)
  resourceId  String?                        // 资源 ID
  
  // 操作详情
  details     Json?                          // 操作详情 (变更内容)
  metadata    Json?                          // 额外元数据
  
  // 操作来源
  source      AuditSource @default(WEB)      // 来源 (WEB, CLI, API)
  ipAddress   String?                        // IP 地址
  userAgent   String?                        // User Agent
  
  // 操作者
  userId      String?
  user        User?    @relation(fields: [userId], references: [id])
  apiKeyId    String?
  apiKey      ApiKey?  @relation(fields: [apiKeyId], references: [id])
  
  createdAt   DateTime @default(now())

  @@index([action])
  @@index([resource, resourceId])
  @@index([userId])
  @@index([createdAt])
}

enum AuditAction {
  CREATE
  UPDATE
  DELETE
  IMPORT
  EXPORT
  LOGIN
  LOGOUT
  ACTIVATE
  VERIFY
}

enum AuditSource {
  WEB
  CLI
  API
  PLUGIN
  SYSTEM
}
```
