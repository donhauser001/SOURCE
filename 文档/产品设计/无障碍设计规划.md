## 一、概述

### 1.1 背景与目标

SOURCE 作为专业色彩管理系统，其核心功能围绕色彩展示、筛选和管理。然而，全球约 **4.5%** 的人口存在不同程度的色觉障碍（色盲或色弱），在男性群体中这一比例高达 **8%**。

本文档旨在规划一套系统化的无障碍设计方案，确保色盲用户能够有效使用 SOURCE 系统的核心功能。

### 1.2 适用标准

本规划遵循以下无障碍设计标准：

| 标准 | 级别 | 说明 |
|------|------|------|
| WCAG 2.1 | AA 级 | Web 内容无障碍指南（主要目标） |
| WCAG 2.1 | AAA 级 | 针对色彩相关功能的增强目标 |
| Section 508 | - | 美国无障碍法规（国际化考量） |
| EN 301 549 | - | 欧盟无障碍标准（国际化考量） |

### 1.3 色觉障碍类型

| 类型 | 医学名称 | 发生率 | 影响描述 |
|------|---------|--------|---------|
| 红色盲 | Protanopia | 1.0% 男性 | 无法感知红色光谱 |
| 红色弱 | Protanomaly | 1.0% 男性 | 红色感知减弱 |
| 绿色盲 | Deuteranopia | 1.0% 男性 | 无法感知绿色光谱 |
| 绿色弱 | Deuteranomaly | 5.0% 男性 | 绿色感知减弱（最常见） |
| 蓝色盲 | Tritanopia | 0.01% | 无法感知蓝色光谱 |
| 蓝色弱 | Tritanomaly | 0.01% | 蓝色感知减弱 |
| 全色盲 | Achromatopsia | 0.003% | 仅能感知明暗 |

## 三、解决方案设计

### 3.1 核心原则

```
┌─────────────────────────────────────────────────────────────┐
│                    无障碍设计四原则                          │
├─────────────────────────────────────────────────────────────┤
│  1. 可感知 (Perceivable)                                    │
│     → 信息必须以用户能感知的方式呈现                         │
│                                                             │
│  2. 可操作 (Operable)                                       │
│     → 界面组件必须可通过多种方式操作                         │
│                                                             │
│  3. 可理解 (Understandable)                                 │
│     → 信息和操作必须易于理解                                 │
│                                                             │
│  4. 健壮性 (Robust)                                         │
│     → 内容必须能被各种辅助技术正确解析                       │
└─────────────────────────────────────────────────────────────┘
```

### 3.4 组件级改造方案

#### 3.4.1 色彩卡片 (ColorCard)

**当前实现：**
```tsx
<div className="w-full h-32 rounded-t-lg" style={{ backgroundColor: rgbColor }} />
```

**无障碍改造：**
```tsx
<div 
  className="w-full h-32 rounded-t-lg relative"
  style={{ backgroundColor: rgbColor }}
  role="img"
  aria-label={`${color.name}，色系：${familyLabel}，Lab值：L${color.labL} a${color.labA} b${color.labB}`}
>
  {/* 纹理叠加层 - 仅在纹理模式下显示 */}
  {accessibilityMode === 'texture' && (
    <svg className="absolute inset-0 w-full h-full opacity-30">
      <rect fill={`url(#${patternId})`} width="100%" height="100%" />
    </svg>
  )}
  
  {/* 色系图标 - 在纹理/高对比度模式下显示 */}
  {['texture', 'high-contrast'].includes(accessibilityMode) && (
    <ColorFamilyIcon family={color.colorFamily} className="absolute top-2 right-2" />
  )}
  
  {/* 强制标签 */}
  {settings.alwaysShowFamilyLabel && (
    <Badge className="absolute bottom-2 left-2">{familyLabel}</Badge>
  )}
</div>
```

```tsx
  {colors.map((color, index) => (
    <ColorCard
      key={color.id}

    />
  ))}

// 键盘导航处理
function handleGridKeyDown(e: KeyboardEvent) {
  const columns = getGridColumns();
  switch (e.key) {
    case 'ArrowRight': moveFocus(1); break;
    case 'ArrowLeft': moveFocus(-1); break;
    case 'ArrowDown': moveFocus(columns); break;
    case 'ArrowUp': moveFocus(-columns); break;
    case 'Home': setFocusedIndex(0); break;
    case 'End': setFocusedIndex(colors.length - 1); break;
  }
}
```

### 3.6 色彩描述系统

为每个颜色提供标准化的文字描述：

```typescript
interface ColorDescription {
  // 基础描述
  brief: string;           // "淡雅的青绿色"
  
  // 详细描述
  detailed: string;        // "一种明度较高、饱和度中等的青绿色调..."
  
  // 联想描述
  associative: string;     // "如同春雨后的竹叶，清新而宁静"
  
  // 技术描述
  technical: string;       // "Lab(65, -12, 8)，位于青绿色区域..."
  
  // 对比参考
  comparison: string;      // "比薄荷绿更深沉，比松石绿更柔和"
}

// 自动生成描述
function generateColorDescription(color: Color): ColorDescription {
  const hue = calculateHue(color.labA, color.labB);
  const chroma = calculateChroma(color.labA, color.labB);
  const lightness = color.labL;
  
  return {
    brief: generateBriefDescription(hue, chroma, lightness),
    detailed: generateDetailedDescription(color),
    associative: color.culturalDescription || generateAssociativeDescription(color),
    technical: generateTechnicalDescription(color),
    comparison: generateComparisonDescription(color),
  };
}
```

```typescript
interface AccessibilityContextType {
  settings: AccessibilitySettings;
  updateSettings: (settings: Partial<AccessibilitySettings>) => void;
  
  // 便捷方法
  isTextureMode: boolean;
  isHighContrastMode: boolean;
  isTextOnlyMode: boolean;

export const AccessibilityProvider: React.FC<PropsWithChildren> = ({ children }) => {
  const [settings, setSettings] = useState<AccessibilitySettings>(() => {
    // 从 localStorage 或用户设置加载
    return loadAccessibilitySettings();
  });
  
  // 监听系统偏好
  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    if (mediaQuery.matches) {
      setSettings(s => ({ ...s, reduceMotion: true }));
    }
  }, []);
  
  // 持久化设置
  useEffect(() => {
    saveAccessibilitySettings(settings);
  }, [settings]);
  
  return (
    <AccessibilityContext.Provider value={contextValue}>
      {children}
    </AccessibilityContext.Provider>
  );
};
```

### 4.3 无障碍 Hook

```typescript
// hooks/use-accessibility.ts

export function useAccessibility() {
  const context = useContext(AccessibilityContext);
  if (!context) {
    throw new Error('useAccessibility must be used within AccessibilityProvider');
  }
  return context;
}

export function useColorAnnouncer() {
  const [announcement, setAnnouncement] = useState('');
  
  const announce = useCallback((message: string, priority: 'polite' | 'assertive' = 'polite') => {
    setAnnouncement('');
    // 强制 DOM 更新以触发屏幕阅读器
    requestAnimationFrame(() => {
      setAnnouncement(message);
    });
  }, []);
  
  return {
    announce,
    AnnouncerRegion: () => (
      <div
        role="status"
        aria-live="polite"
        aria-atomic="true"
        className="sr-only"
      >
        {announcement}
      </div>
    ),
  };
}

export function useReducedMotion() {
  const [prefersReducedMotion, setPrefersReducedMotion] = useState(false);
  
  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    setPrefersReducedMotion(mediaQuery.matches);
    
    const handler = (e: MediaQueryListEvent) => setPrefersReducedMotion(e.matches);
    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, []);
  
  return prefersReducedMotion;
}
```

```typescript
import { test, expect } from '@playwright/test';

const COLORBLIND_TYPES = [
  'protanopia',
  'deuteranopia', 
  'tritanopia',
  'achromatopsia',
];

test.describe('色盲模拟测试', () => {
  for (const type of COLORBLIND_TYPES) {
    test(`${type} 模式下色系应可区分`, async ({ page }) => {
      await page.goto('/colors');
      
      // 启用纹理模式
      await page.evaluate(() => {
        localStorage.setItem('accessibility-settings', JSON.stringify({
          mode: 'texture',
          alwaysShowFamilyLabel: true,
        }));
      });
      await page.reload();
      
      // 应用色盲模拟 CSS 滤镜
      await page.addStyleTag({
        content: getColorBlindFilter(type),
      });
      
      // 截图对比
      await expect(page).toHaveScreenshot(`colorblind-${type}.png`);
      
      // 验证纹理和标签可见
      const patterns = page.locator('[data-pattern]');
      expect(await patterns.count()).toBeGreaterThan(0);
      
      const labels = page.locator('[data-color-family-label]');
      expect(await labels.count()).toBeGreaterThan(0);
    });
  }
});
```
