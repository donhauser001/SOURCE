# SOURCE 相机取色校准流程设计

> **核心问题**：手机相机受白平衡、色温、环境光影响极大，直接取色误差可能导致"色彩幻觉"。  
> **解决方案**：灰卡校准 + 环境光检测 + 置信度提示，让用户知道"这个结果有多可信"。

## 三、灰卡校准模块（核心）

### 3.1 校准原理

```
理论值（D50标准光源下）：
  灰卡中灰: R=128, G=128, B=128 (18%灰)
  灰卡白点: R=243, G=243, B=243 (90%白)
  灰卡黑点: R=52,  G=52,  B=52  (3.5%黑)

实际拍摄值（受环境光影响）：
  灰卡中灰: R=142, G=135, B=118  ← 偏暖
  
校准补偿矩阵：
  R_corrected = R_captured × (128/142)
  G_corrected = G_captured × (128/135)  
  B_corrected = B_captured × (128/118)
```

### 3.4 校准算法实现

```typescript
// 灰卡校准核心逻辑
interface CalibrationResult {
  success: boolean;
  colorMatrix: ColorMatrix;     // 3x3 色彩校正矩阵
  tempOffset: number;           // 色温偏移量 (K)
  exposureComp: number;         // 曝光补偿建议
  confidence: number;           // 校准置信度 (0-1)
  error?: string;
}

interface GrayCardPatch {
  name: 'black' | 'midGray' | 'white';
  referenceRGB: RGB;            // 标准参考值
  referenceLab: Lab;            // 标准 Lab 值
  capturedRGB: RGB;             // 实际拍摄值
  deviation: number;            // 偏差
}

class GrayCardCalibrator {
  // 标准参考值 (D50 标准光源)
  private static readonly REFERENCE_PATCHES: GrayCardPatch[] = [
    { name: 'black',   referenceRGB: { r: 52, g: 52, b: 52 },     referenceLab: { l: 20, a: 0, b: 0 } },
    { name: 'midGray', referenceRGB: { r: 128, g: 128, b: 128 },  referenceLab: { l: 50, a: 0, b: 0 } },
    { name: 'white',   referenceRGB: { r: 243, g: 243, b: 243 },  referenceLab: { l: 96, a: 0, b: 0 } },
  ];

  /**
   * 执行灰卡校准
   */
  async calibrate(imageData: ImageData): Promise<CalibrationResult> {
    // 1. 检测灰卡位置
    const patches = await this.detectGrayCardPatches(imageData);
    if (!patches) {
      return { success: false, error: '未检测到灰卡，请重新拍摄' };
    }

    // 2. 提取各色块的实际 RGB 值
    const capturedPatches = this.extractPatchColors(imageData, patches);

    // 3. 计算色彩校正矩阵
    const colorMatrix = this.calculateColorMatrix(capturedPatches);

    // 4. 估算色温偏移
    const tempOffset = this.estimateColorTempOffset(capturedPatches);

    // 5. 计算校准置信度
    const confidence = this.calculateConfidence(capturedPatches);

    return {
      success: true,
      colorMatrix,
      tempOffset,
      exposureComp: this.calculateExposureComp(capturedPatches),
      confidence,
    };
  }

  /**
   * 应用校准补偿
   */
  applyCalibration(rgb: RGB, calibration: CalibrationResult): RGB {
    if (!calibration.success) return rgb;

    const { colorMatrix } = calibration;
    
    // 应用 3x3 色彩矩阵
    return {
      r: clamp(colorMatrix[0][0] * rgb.r + colorMatrix[0][1] * rgb.g + colorMatrix[0][2] * rgb.b, 0, 255),
      g: clamp(colorMatrix[1][0] * rgb.r + colorMatrix[1][1] * rgb.g + colorMatrix[1][2] * rgb.b, 0, 255),
      b: clamp(colorMatrix[2][0] * rgb.r + colorMatrix[2][1] * rgb.g + colorMatrix[2][2] * rgb.b, 0, 255),
    };
  }

  /**
   * 计算 3x3 色彩校正矩阵
   * 使用最小二乘法拟合
   */
  private calculateColorMatrix(patches: GrayCardPatch[]): ColorMatrix {
    // 简化版：对角矩阵（独立通道增益）
    const midGray = patches.find(p => p.name === 'midGray')!;
    const ref = REFERENCE_PATCHES.find(p => p.name === 'midGray')!;

    const rGain = ref.referenceRGB.r / midGray.capturedRGB.r;
    const gGain = ref.referenceRGB.g / midGray.capturedRGB.g;
    const bGain = ref.referenceRGB.b / midGray.capturedRGB.b;

    return [
      [rGain, 0, 0],
      [0, gGain, 0],
      [0, 0, bGain],
    ];
  }

  /**
   * 估算色温偏移
   */
  private estimateColorTempOffset(patches: GrayCardPatch[]): number {
    const midGray = patches.find(p => p.name === 'midGray')!;
    const { r, g, b } = midGray.capturedRGB;
    
    // 简化的色温估算：基于 R/B 比值
    const rbRatio = r / b;
    
    // 标准 D50 (5000K) 下 R/B ≈ 1.0
    // R/B > 1 → 偏暖 (< 5000K)
    // R/B < 1 → 偏冷 (> 5000K)
    
    const tempOffset = (rbRatio - 1) * 1500; // 粗略映射
    return Math.round(tempOffset);
  }

  /**
   * 计算校准置信度
   */
  private calculateConfidence(patches: GrayCardPatch[]): number {
    // 基于各色块的色度偏差计算置信度
    // a*, b* 应该接近 0（中性灰）
    
    let totalChromaError = 0;
    for (const patch of patches) {
      const lab = rgbToLab(patch.capturedRGB.r, patch.capturedRGB.g, patch.capturedRGB.b);
      const chromaError = Math.sqrt(lab.a * lab.a + lab.b * lab.b);
      totalChromaError += chromaError;
    }
    
    const avgChromaError = totalChromaError / patches.length;
    
    // 色度误差越小，置信度越高
    // avgChromaError = 0 → confidence = 1.0
    // avgChromaError = 20 → confidence ≈ 0.5
    const confidence = Math.exp(-avgChromaError / 20);
    
    return Math.round(confidence * 100) / 100;
  }
}
```

### 4.2 置信度与提示逻辑

```typescript
interface ColorMatchResult {
  capturedLab: Lab;               // 取色 Lab 值
  calibrationApplied: boolean;    // 是否应用了校准
  calibrationConfidence?: number; // 校准置信度
  matches: SimilarColor[];        // 匹配结果
  overallConfidence: number;      // 综合置信度
  warnings: string[];             // 警告信息
}

function generateConfidenceWarnings(result: ColorMatchResult): string[] {
  const warnings: string[] = [];

  // 1. 校准置信度警告
  if (!result.calibrationApplied) {
    warnings.push('未使用灰卡校准，结果可能存在偏差');
  } else if (result.calibrationConfidence && result.calibrationConfidence < 0.7) {
    warnings.push('校准置信度较低，建议重新拍摄灰卡');
  }

  // 2. 匹配质量警告
  if (result.matches.length === 0) {
    warnings.push('未找到相似色彩，该颜色可能不在 SOURCE 数据库中');
  } else if (result.matches[0].deltaE > 5) {
    warnings.push('最佳匹配色差较大 (ΔE > 5)，建议人工确认');
  }

  // 3. 极端颜色警告
  if (result.capturedLab.l < 15 || result.capturedLab.l > 95) {
    warnings.push('取色区域过暗/过亮，可能影响准确度');
  }

  return warnings;
}

// 置信度分级展示
function getConfidenceLevel(confidence: number): {
  level: 'excellent' | 'good' | 'fair' | 'poor';
  label: string;
  color: string;
} {
  if (confidence >= 0.9) return { level: 'excellent', label: '极高', color: '#22c55e' };
  if (confidence >= 0.7) return { level: 'good', label: '良好', color: '#3b82f6' };
  if (confidence >= 0.5) return { level: 'fair', label: '中等', color: '#f59e0b' };
  return { level: 'poor', label: '较低', color: '#ef4444' };
}
```

### 4.3 ΔE 匹配等级说明

### 5.2 软件补偿策略

```typescript
// 无灰卡时的软件估算补偿
class AutoColorCorrection {
  /**
   * 基于图像内容的自动白平衡
   */
  autoWhiteBalance(imageData: ImageData): ColorMatrix {
    // 1. 灰色世界假设：假设图像平均颜色应为中性灰
    const avgRGB = this.calculateAverageRGB(imageData);
    
    // 2. 计算增益
    const avgGray = (avgRGB.r + avgRGB.g + avgRGB.b) / 3;
    const rGain = avgGray / avgRGB.r;
    const gGain = avgGray / avgRGB.g;
    const bGain = avgGray / avgRGB.b;
    
    return [
      [rGain, 0, 0],
      [0, gGain, 0],
      [0, 0, bGain],
    ];
  }

  /**
   * 高光区域白平衡（假设最亮点为白色）
   */
  highlightWhiteBalance(imageData: ImageData): ColorMatrix {
    // 寻找图像中最亮的区域，假设为白色
    const highlight = this.findHighlightRegion(imageData);
    
    // 计算到纯白的校正
    const rGain = 255 / highlight.r;
    const gGain = 255 / highlight.g;
    const bGain = 255 / highlight.b;
    
    return [
      [rGain, 0, 0],
      [0, gGain, 0],
      [0, 0, bGain],
    ];
  }
}
```

## 六、性能优化

### 6.1 相似色搜索优化

```typescript
// 问题：全表扫描 + ΔE2000 计算开销大
// 解决方案：多级过滤

class OptimizedColorMatcher {
  /**
   * 优化后的相似色搜索
   */
  async findSimilarColors(
    targetLab: Lab, 
    limit: number = 5
  ): Promise<SimilarColor[]> {
    // 第一级：Lab 范围粗筛（利用数据库索引）
    const candidates = await prisma.color.findMany({
      where: {
        trueSource: {
          labL: { gte: targetLab.l - 20, lte: targetLab.l + 20 },
          labA: { gte: targetLab.a - 30, lte: targetLab.a + 30 },
          labB: { gte: targetLab.b - 30, lte: targetLab.b + 30 },
        },
        status: 'ACTIVE',
      },
      select: {
        id: true,
        colorId: true,
        name: true,
        trueSource: { select: { labL: true, labA: true, labB: true } },
      },
    });

    // 第二级：ΔE2000 精确计算（内存中）
    const results = candidates
      .map(color => ({
        ...color,
        deltaE: deltaE2000(targetLab, {
          l: color.trueSource.labL,
          a: color.trueSource.labA,
          b: color.trueSource.labB,
        }),
      }))
      .sort((a, b) => a.deltaE - b.deltaE)
      .slice(0, limit);

    return results;
  }
}

// 索引建议
// CREATE INDEX idx_color_lab ON "TrueSource" ("labL", "labA", "labB");
```

### 6.2 结果缓存策略

```typescript
// Redis 缓存热门查询
class ColorMatchCache {
  private redis: Redis;
  private ttl = 3600; // 1小时

  /**
   * 将 Lab 值量化为缓存键
   * L: 精度 1, a/b: 精度 2
   */
  private labToKey(lab: Lab): string {
    const l = Math.round(lab.l);
    const a = Math.round(lab.a / 2) * 2;
    const b = Math.round(lab.b / 2) * 2;
    return `color:match:${l}:${a}:${b}`;
  }

  async getCachedMatch(lab: Lab): Promise<SimilarColor[] | null> {
    const key = this.labToKey(lab);
    const cached = await this.redis.get(key);
    return cached ? JSON.parse(cached) : null;
  }

  async setCachedMatch(lab: Lab, results: SimilarColor[]): Promise<void> {
    const key = this.labToKey(lab);
    await this.redis.setex(key, this.ttl, JSON.stringify(results));
  }
}
```

## 八、总结

### 核心设计原则

1. **诚实的不确定性**：始终告诉用户"这个结果有多可信"，而不是假装100%准确
2. **渐进式引导**：鼓励使用灰卡校准，但不强制，快速模式也能用
3. **可追溯的证据**：匹配结果链接到完整的色彩身份证，用户可自行验证
